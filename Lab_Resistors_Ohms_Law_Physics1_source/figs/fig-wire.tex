\documentclass[border=2pt]{standalone}
\usepackage{labstyle}
\begin{document}
\begin{tikzpicture}[lab]
  % a copper-coloured wire: length L (cm on paper) and diameter d (cm on paper); dots = atoms that scatter the electrons
  \newcommand\wire[5]{% #1 x0, #2 y0, #3 L, #4 d, #5 number of dots
    \pgfmathsetmacro\rr{#4/2}\pgfmathsetmacro\ex{0.22*#4}
    \shade[top color=copperhi, bottom color=copperlo, middle color=copper]
      (#1,#2-\rr) -- (#1+#3,#2-\rr) arc[start angle=-90, end angle=90, x radius=\ex, y radius=\rr] -- (#1,#2+\rr) -- cycle;
    \begin{scope}
      \clip (#1,#2-\rr) rectangle (#1+#3,#2+\rr);
      \pgfmathsetseed{#5}
      \foreach \k in {1,...,#5} {
        \pgfmathsetmacro\px{#1+0.06+rnd*(#3-0.12)}\pgfmathsetmacro\py{#2+(rnd-0.5)*1.7*\rr}
        \fill[copperlo!65!black, opacity=0.75] (\px,\py) circle (0.022);
      }
    \end{scope}
    \fill[white, opacity=0.22] (#1,#2+0.32*\rr) rectangle (#1+#3,#2+0.55*\rr);
    \draw[copperlo, line width=0.5pt] (#1,#2-\rr) -- (#1+#3,#2-\rr) arc[start angle=-90, end angle=90, x radius=\ex, y radius=\rr] -- (#1,#2+\rr);
    \fill[copper!55!copperhi] (#1,#2) ellipse[x radius=\ex, y radius=\rr];
    \draw[copperlo, line width=0.5pt] (#1,#2) ellipse[x radius=\ex, y radius=\rr];
  }
  \newcommand\card[6]{% #1 x, #2 y (top), #3 rho, #4 L, #5 A, #6 R
    \fill[soft, rounded corners=2.5mm] (#1,#2) rectangle ++(4.6,-1.9);
    \node[anchor=north west, inner sep=0pt, align=left, font=\small] at (#1+0.3,#2-0.25)
      {\begin{tabular}{@{}l@{\;}l@{}}
         \(\rho\) & = \qty{#3}{\ohm\centi\metre}\\
         \(L\)    & = \qty{#4}{\centi\metre}\\
         \(A\)    & = \qty{#5}{\centi\metre\squared}
       \end{tabular}};
    \node[anchor=north east, inner sep=0pt, align=right] at (#1+4.35,#2-0.32)
      {{\footnotesize\color{muted}resistans}\\[1pt]{\fsSemi\Large\color{navy}\qty{#6}{\ohm}}};
  }
  % ---- wire A: short and thick
  \node[anchor=west, font=\fsSemi, text=navy] at (0,1.25) {Tråd A};
  \wire{0.25}{0}{2.50}{0.90}{40}
  \draw[|<->|, muted, line width=0.6pt] (0.25,-0.72) -- (2.75,-0.72) node[midway, below, tag]{\(L = \qty{10.00}{\centi\metre}\)};
  \draw[callout] (0.25,0.28) -- (-0.15,0.80) node[anchor=south, tag, inner sep=1pt]{\(A\)};
  \card{5.7}{1.0}{0.50}{10.00}{7.50}{0.667}
  % ---- wire B: long and thin
  \node[anchor=west, font=\fsSemi, text=navy] at (0,-1.85) {Tråd B};
  \wire{0.25}{-2.95}{4.72}{0.55}{47}
  \draw[|<->|, muted, line width=0.6pt] (0.25,-3.52) -- (4.97,-3.52) node[midway, below, tag]{\(L = \qty{18.86}{\centi\metre}\)};
  \draw[callout] (0.25,-2.80) -- (-0.15,-2.30) node[anchor=south, tag, inner sep=1pt]{\(A\)};
  \card{5.7}{-2.05}{0.50}{18.86}{2.76}{3.42}
  % arrow between the rows
\end{tikzpicture}
\end{document}
